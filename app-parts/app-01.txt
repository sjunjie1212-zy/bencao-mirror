function esc(s=''){ return String(s).replace(/[&<>'"]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;',"'":'&#39;','"':'&quot;'}[c])); }
function img(src,alt,extra=''){
  if(String(src).startsWith('sprite:')){
    const [,herb,cell]=String(src).split(':');
    return `<span class="bencao-sprite" data-sprite="${herb}" data-cell="${cell}" role="img" aria-label="${esc(alt)}"></span>`;
  }
  return `<img src="${src}" alt="${esc(alt)}" loading="lazy" ${extra} onerror="this.classList.add('image-failed')">`;
}
const SPRITE_POS={0:'0% 0%',1:'50% 0%',2:'100% 0%',3:'0% 100%',4:'50% 100%',5:'100% 100%'};
const spriteCache=new Map();
async function spriteDataUrl(herb){
  if(spriteCache.has(herb)) return spriteCache.get(herb);
  const promise=fetch(`./assets/${herb}.b64?v=${VERSION}`,{cache:'force-cache'})
    .then(async r=>{ if(!r.ok) throw new Error(`sprite ${herb}`); return `data:image/webp;base64,${(await r.text()).trim()}`; });
  spriteCache.set(herb,promise);
  return promise;
}
async function hydrateSprites(root=document){
  const nodes=[...root.querySelectorAll('.bencao-sprite:not([data-ready])')];
  await Promise.all(nodes.map(async el=>{
    try{
      const herb=el.dataset.sprite;
      const url=await spriteDataUrl(herb);
      el.style.backgroundImage=`url(${url})`;
      el.style.backgroundPosition=SPRITE_POS[Number(el.dataset.cell)]||'0% 0%';
      el.dataset.ready='1';
    }catch(_){ el.classList.add('image-failed'); }
  }));
}

function rgbChannels(hex){
  const v=String(hex||'').replace('#','').trim();
  const full=v.length===3?v.split('').map(x=>x+x).join(''):v;
  if(!/^[0-9a-fA-F]{6}$/.test(full)) return [120,120,120];
  return [parseInt(full.slice(0,2),16),parseInt(full.slice(2,4),16),parseInt(full.slice(4,6),16)];
}
function rgba(hex,a){ const [r,g,b]=rgbChannels(hex); return `rgba(${r},${g},${b},${a})`; }
